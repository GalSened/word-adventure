**ScreenRouter.jsx:**
Create a switching component that maps `gameState` to screen components:

```jsx
import { AnimatePresence } from 'framer-motion';

const SCREEN_MAP = {
  start: StartScreen,
  map: MapScreen,
  playing: PlayingScreen,
  levelComplete: ResultScreen,  // isSuccess=true
  gameOver: ResultScreen,        // isSuccess=false
  store: StoreScreen,
  inventory: InventoryScreen,
  memory: MemoryScreen,
  petWalking: PetWalkingScreen,
  avatar: AvatarScreen,
};
```
